import random,discord
from discord.ui import View,button

high={}

class Game(View):
 def __init__(self,u):
  super().__init__(timeout=300);self.u=u;self.s=0;self.a=[[0]*4 for _ in range(4)]
  self.add();self.add()

 def add(self):
  z=[(r,c) for r in range(4) for c in range(4) if not self.a[r][c]]
  if z:
   r,c=random.choice(z);self.a[r][c]=4 if random.random()<.1 else 2

 def move(self,d):
  old=[r[:] for r in self.a]
  for r in range(4):
   x=self.a[r][:] if d in "LR" else [self.a[c][r] for c in range(4)]
   x=[v for v in x if v]
   if d in "RD":x=x[::-1]
   y=[];i=0
   while i<len(x):
    if i+1<len(x) and x[i]==x[i+1]:
     y.append(x[i]*2);self.s+=x[i]*2;i+=2
    else:y.append(x[i]);i+=1
   y += [0]*(4-len(y))
   if d in "RD":y=y[::-1]
   if d in "LR":self.a[r]=y
   else:
    for c in range(4):self.a[c][r]=y[c]
  if old!=self.a:self.add();return True
  return False

 def over(self):
  if any(0 in r for r in self.a):return False
  for r in range(4):
   for c in range(4):
    if c<3 and self.a[r][c]==self.a[r][c+1]:return False
    if r<3 and self.a[r][c]==self.a[r+1][c]:return False
  return True

 def board(self):
  colors={0:"⬛",2:"🟨",4:"🟧",8:"🟥",16:"🟪",
          32:"🟦",64:"🟩",128:"🟫",256:"🔵",
          512:"🟣",1024:"🔴",2048:"🏆"}
  return "\n".join(" ".join(f"{colors.get(x,'⭐')}{x or ''}" for x in r) for r in self.a)

 async def play(self,i,d):
  if i.user.id!=self.u:return
  if not self.move(d):
   return await i.response.send_message("❌ **No moves that way!**",ephemeral=True)

  if any(2048 in r for r in self.a):
   high[self.u]=max(high.get(self.u,0),self.s);self.clear()
   return await i.response.edit_message(
    content=f"🏆 **YOU REACHED 2048!**\n\n{self.board()}\n\n"
            f"⭐ Score: **{self.s:,}**\n🏅 Best: **{high[self.u]:,}**",view=self)

  if self.over():
   high[self.u]=max(high.get(self.u,0),self.s);self.clear()
   return await i.response.edit_message(
    content=f"💀 **GAME OVER!**\n\n{self.board()}\n\n"
            f"⭐ Score: **{self.s:,}**\n🏅 Best: **{high[self.u]:,}**\n\n"
            "Press 🔄 to play again!",view=self)

  await i.response.edit_message(
   content=f"🧩 **2048**\n\n{self.board()}\n\n"
           f"⭐ Score: **{self.s:,}**  •  🏅 Best: **{high.get(self.u,0):,}**",
   view=self)

 @button(label="⬆️",style=discord.ButtonStyle.blurple,row=0)
 async def up(self,i,b):await self.play(i,"U")

 @button(label="⬅️",style=discord.ButtonStyle.blurple,row=1)
 async def left(self,i,b):await self.play(i,"L")

 @button(label="⬇️",style=discord.ButtonStyle.blurple,row=1)
 async def down(self,i,b):await self.play(i,"D")

 @button(label="➡️",style=discord.ButtonStyle.blurple,row=1)
 async def right(self,i,b):await self.play(i,"R")

 @button(label="🔄 Restart",style=discord.ButtonStyle.green,row=2)
 async def restart(self,i,b):
  if i.user.id!=self.u:return
  self.a=[[0]*4 for _ in range(4)];self.s=0
  self.add();self.add()
  for x in self.children:x.disabled=False
  await i.response.edit_message(
   content=f"🧩 **2048**\n\n{self.board()}\n\n"
           f"⭐ Score: **0**  •  🏅 Best: **{high.get(self.u,0):,}**",view=self)
